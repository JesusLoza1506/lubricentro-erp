include!("vehiculo_tests.rs");
use lubricentro_erp_lib::services::ordenes_service::OrdenesService;
use lubricentro_erp_lib::dto::orden_trabajo_dto::CrearOrdenTrabajoDto;

fn seed_finished(oil: bool, material: bool) -> Connection {
    let c = setup_test_db();
    c.execute("INSERT INTO servicios(codigo,nombre,precio_venta_cent,es_cambio_aceite) VALUES('TMP','Servicio prueba',1500,?1)", [oil as i32]).unwrap();
    c.execute_batch("INSERT INTO ordenes_trabajo(id_operacion,codigo_ot,id_vehiculo,id_cliente_atendido,placa_snapshot,cliente_snapshot,id_creador,estado,kilometraje_ingreso) VALUES('TMP','OT-TMP',1,2,'ABC123','{\"nombre\":\"Cliente historico\",\"tipo_documento\":\"DNI\",\"numero_documento\":\"12345678\"}',1,'EN_ESPERA',50000); UPDATE ordenes_trabajo SET estado='EN_PROCESO',id_mecanico=3,id_zanja_ejecucion=1,fecha_inicio=datetime('now','-5 hours') WHERE id_ot=1;").unwrap();
    c.execute("INSERT INTO detalle_ot_servicios(id_operacion,id_ot,id_servicio,codigo_snapshot,nombre_snapshot,cantidad,precio_unitario_cent,es_cambio_aceite,id_usuario) VALUES('TMP-S',1,1,'TMP','Servicio prueba',1,1500,?1,1)", [oil as i32]).unwrap();
    if material { c.execute_batch("INSERT INTO productos(codigo,nombre,categoria,unidad_medida,stock_actual,stock_minimo,valor_stock_micro,precio_venta_cent,activo) VALUES('TMP-P','Aceite mineral','ACEITE','UND',0,2,0,5000,1); INSERT INTO detalle_ot_productos(id_operacion,id_ot,id_producto,codigo_snapshot,nombre_snapshot,unidad_medida,cantidad,precio_unitario_cent,costo_unitario_micro,valor_consumido_micro,id_usuario) VALUES('TMP-P',1,1,'TMP-P','Aceite mineral','UND',1,5000,10000,10000,1);").unwrap(); }
    c.execute("UPDATE ordenes_trabajo SET estado='FINALIZADO',id_mecanico_terminal=3,kilometraje_salida=50010,tipo_aceite=?1,intervalo_aceite_km=?2,resumen_publico='Resumen publico',fecha_fin=datetime('now','-5 hours') WHERE id_ot=1", rusqlite::params![if oil {Some("SINTETICO")} else {None}, if oil {Some(10000)} else {None}]).unwrap();
    c
}
#[test]
fn quinta_historia_devuelve_cliente_snapshot() {
    let c=seed_finished(false,false);
    let f=ClienteService::obtener_ficha_vehicular_completa(&c,"ABC123",Some("ADMINISTRADOR")).unwrap();
    let json=serde_json::to_string(&f.historial[0]).unwrap();
    assert!(json.contains("Cliente historico"),"DTO real: {}",json);
}
#[test]
fn quinta_material_no_inventa_cambio() {
    let c=seed_finished(false,true);
    let stored:Option<i64>=c.query_row("SELECT km_proximo FROM ordenes_trabajo WHERE id_ot=1",[],|r|r.get(0)).unwrap();
    assert_eq!(stored,None);
    let f=ClienteService::obtener_ficha_vehicular_completa(&c,"ABC123",Some("ADMINISTRADOR")).unwrap();
    assert_eq!(f.historial[0].proximo_kilometraje,stored);
}

#[test]
fn quinta_login_mecanico_ficha_solo_lectura() {
    use lubricentro_erp_lib::services::auth_service::AuthService;
    use lubricentro_erp_lib::dto::usuario_dto::LoginRequestDto;
    let c=setup_test_db();
    let hash=AuthService::hash_password("Taller2026Clave").unwrap();
    c.execute("UPDATE usuarios SET password_hash=?1 WHERE id_usuario=3",[hash]).unwrap();
    let (r,_) = AuthService::login(&c,LoginRequestDto{username:"meca".into(),password:"Taller2026Clave".into()}).unwrap();
    let p=r.permisos.iter().find(|p|p.modulo=="FICHA_VEHICULAR").unwrap();
    assert!(p.puede_ver);
    assert!(!p.puede_crear && !p.puede_editar && !p.puede_eliminar);
}
