fn seed_finished(oil: bool, material: bool) -> Connection {
    seed_finished_snapshot(oil, material, r#"{"nombre":"Cliente historico","documento":"12345678","tipo_documento":"DNI"}"#)
}
fn seed_finished_snapshot(oil: bool, material: bool, snapshot: &str) -> Connection {
    let c = setup_test_db();
    c.execute("INSERT INTO servicios(codigo,nombre,precio_venta_cent,es_cambio_aceite) VALUES('TMP','Servicio prueba',1500,?1)", [oil as i32]).unwrap();
    c.execute("INSERT INTO ordenes_trabajo(id_operacion,codigo_ot,id_vehiculo,id_cliente_atendido,placa_snapshot,cliente_snapshot,id_creador,estado,kilometraje_ingreso) VALUES('TMP','OT-TMP',1,2,'ABC123',?1,1,'EN_ESPERA',50000)",[snapshot]).unwrap();
    c.execute_batch("UPDATE ordenes_trabajo SET estado='EN_PROCESO',id_mecanico=3,id_zanja_ejecucion=1,fecha_inicio=datetime('now','-5 hours') WHERE id_ot=1;").unwrap();
    c.execute("INSERT INTO detalle_ot_servicios(id_operacion,id_ot,id_servicio,codigo_snapshot,nombre_snapshot,cantidad,precio_unitario_cent,es_cambio_aceite,id_usuario) VALUES('TMP-S',1,1,'TMP','Servicio prueba',1,1500,?1,1)", [oil as i32]).unwrap();
    if material { c.execute_batch("INSERT INTO productos(codigo,nombre,categoria,unidad_medida,stock_actual,stock_minimo,valor_stock_micro,precio_venta_cent,activo) VALUES('TMP-P','Aceite mineral','ACEITE','UND',0,2,0,5000,1); INSERT INTO detalle_ot_productos(id_operacion,id_ot,id_producto,codigo_snapshot,nombre_snapshot,unidad_medida,cantidad,precio_unitario_cent,costo_unitario_micro,valor_consumido_micro,id_usuario) VALUES('TMP-P',1,1,'TMP-P','Aceite mineral','UND',1,5000,10000,10000,1);").unwrap(); }
    c.execute("UPDATE ordenes_trabajo SET estado='FINALIZADO',id_mecanico_terminal=3,kilometraje_salida=50010,tipo_aceite=?1,intervalo_aceite_km=?2,resumen_publico='Resumen publico',fecha_fin=datetime('now','-5 hours') WHERE id_ot=1", rusqlite::params![if oil {Some("SINTETICO")} else {None}, if oil {Some(10000)} else {None}]).unwrap();
    c
}

    use lubricentro_erp_lib::dto::usuario_dto::LoginRequestDto;
    let c=setup_test_db();
    let hash=AuthService::hash_password("Taller2026Clave").unwrap();
    c.execute("UPDATE usuarios SET password_hash=?1 WHERE id_usuario=3",[hash]).unwrap();
    let (r,_) = AuthService::login(&c,LoginRequestDto{username:"meca".into(),password:"Taller2026Clave".into()}).unwrap();
    let p=r.permisos.iter().find(|p|p.modulo=="FICHA_VEHICULAR").unwrap();

}

fn sexta_cliente_historico_no_cambia_al_editar_actual() {
    let c=seed_finished(false,false);
    let original=ClienteService::obtener_ficha_vehicular_completa(&c,"ABC123",Some("ADMINISTRADOR")).unwrap();
    c.execute("UPDATE clientes SET nombre_razon_social='Cliente cambiado',numero_documento='87654321' WHERE id_cliente=2",[]).unwrap();
    let despues=ClienteService::obtener_ficha_vehicular_completa(&c,"ABC123",Some("ADMINISTRADOR")).unwrap();
    assert_eq!(despues.historial[0].nombre_cliente,original.historial[0].nombre_cliente,"El nombre historico cambio al editar cliente actual");
    assert_eq!(despues.historial[0].documento_cliente,original.historial[0].documento_cliente);
}

#[test]
fn octava_candidatos_mas_antiguo_primero() {
    let c=setup_test_db();
    c.execute_batch("INSERT INTO vehiculos(placa,id_cliente,marca,modelo,tipo_motor,kilometraje_actual) VALUES('DEF456',2,'Toyota','Yaris','GASOLINA',50000); INSERT INTO servicios(codigo,nombre,precio_venta_cent,es_cambio_aceite) VALUES('CAMBIO','Cambio aceite',1500,1);").unwrap();
    for (id,placa,dias) in [(1,"ABC123",200),(2,"DEF456",190)] {
        let op=format!("CAND-{}",id); let fecha=format!("-{} days",dias);
        c.execute("INSERT INTO ordenes_trabajo(id_operacion,codigo_ot,id_vehiculo,id_cliente_atendido,placa_snapshot,cliente_snapshot,id_creador,estado,kilometraje_ingreso,fecha_ingreso) VALUES(?1,?1,?2,2,?3,'{\"nombre\":\"Juan Perez\"}',1,'EN_ESPERA',50000,datetime('now','-5 hours',?4))",rusqlite::params![op,id,placa,fecha]).unwrap();
        c.execute("UPDATE ordenes_trabajo SET estado='EN_PROCESO',id_mecanico=3,id_zanja_ejecucion=1,fecha_inicio=fecha_ingreso WHERE id_ot=?1",[id]).unwrap();
        c.execute("INSERT INTO detalle_ot_servicios(id_operacion,id_ot,id_servicio,codigo_snapshot,nombre_snapshot,cantidad,precio_unitario_cent,es_cambio_aceite,id_usuario) VALUES(?1,?2,1,'CAMBIO','Cambio aceite',1,1500,1,1)",rusqlite::params![format!("DET-{}",id),id]).unwrap();
        c.execute("UPDATE ordenes_trabajo SET estado='FINALIZADO',id_mecanico_terminal=3,kilometraje_salida=50000,tipo_aceite='MINERAL',intervalo_aceite_km=5000,resumen_publico='Cambio finalizado',fecha_fin=fecha_inicio WHERE id_ot=?1",[id]).unwrap();
    }
    let lista=ClienteService::obtener_proximos_cambios_aceite(&c,Some("ADMINISTRADOR")).unwrap();
    assert_eq!(lista.iter().map(|x|x.id_ot).collect::<Vec<_>>(),vec![1,2],"Candidato de 200 dias debe ir antes que el de 190 dias");
}
