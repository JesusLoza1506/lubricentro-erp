import { render, screen, fireEvent, cleanup, waitFor } from '@testing-library/react';
import { afterEach, beforeEach, expect, it, vi } from 'vitest';
import { ModalCliente } from '../components/ModalCliente';
import { ModalVehiculo } from '../components/ModalVehiculo';
import { ModalDetalleOT } from '../components/ModalDetalleOT';
import { FichaVehicularHistorial } from '../components/FichaVehicularHistorial';

import { vehiculoService } from '../../../services/vehiculoService';
import { ModalCrearOT } from '../../ordenes/components/ModalCrearOT';
import { ordenesService } from '../../../services/ordenesService';
const auth = vi.hoisted(() => ({ rol: 'ADMINISTRADOR' }));
vi.mock('../../../context/AuthContext', () => ({ useAuth: () => ({ tienePermiso: () => true, esAdmin: auth.rol === 'ADMINISTRADOR', usuario: { id_usuario: 1, rol: auth.rol } }) }));
vi.mock('../../../services/vehiculoService', () => ({ vehiculoService: { listarVehiculos:vi.fn(), obtenerVehiculoPorPlaca:vi.fn(), obtenerFichaVehicularCompleta: vi.fn(), obtenerProximosCambiosAceite: vi.fn().mockResolvedValue([]), registrarVehiculo: vi.fn(), actualizarVehiculo: vi.fn(), cambiarEstadoVehiculo: vi.fn() } }));
vi.mock('../../../services/ordenesService', () => ({ ordenesService: { listarMecanicos:vi.fn(), listarOrdenesTrabajo: vi.fn().mockResolvedValue([]), crearOrdenTrabajo: vi.fn() } }));
const vehicle: any = { id_vehiculo: 1, id_cliente: 2, placa: 'ABC123', marca: 'Toyota', modelo: 'Yaris', anio: 2020, tipo_motor: 'GASOLINA', kilometraje_actual: 50000, activo: 1, nombre_cliente: 'Juan Perez' };
const ot: any = { id_ot: 1, codigo_ot: 'OT-001', placa: 'ABC123', id_mecanico: 3, estado: 'FINALIZADO', kilometraje_ingreso: 50010, proximo_kilometraje: null, tipo_aceite: null, fecha_ingreso: '2026-10-18T12:00:00', detalles: [] };
beforeEach(() => { vi.clearAllMocks(); auth.rol = 'ADMINISTRADOR'; vi.mocked(vehiculoService.listarVehiculos).mockResolvedValue([vehicle]); vi.mocked(ordenesService.listarMecanicos).mockResolvedValue([{id_usuario:3,nombre_completo:'Mecanico prueba'}] as any); });
afterEach(cleanup);
it('recepcion normal envia espera sin zanja ni aceite final', async () => {
  const submit=vi.fn().mockResolvedValue(undefined);
  render(<ModalCrearOT idMecanicoDefault={0} placaInicial="ABC123" onClose={vi.fn()} onSubmit={submit}/>);
  const button=screen.getByRole('button',{name:'Crear OT'});
  await waitFor(()=>expect((button as HTMLButtonElement).disabled).toBe(false));
  fireEvent.change(screen.getByPlaceholderText('Ej. 65200'),{target:{value:'50010'}});
  fireEvent.click(button);
  await waitFor(()=>expect(submit).toHaveBeenCalledOnce());
  expect(submit.mock.calls[0][0]).toMatchObject({id_mecanico:0,tipo_aceite:''});
  expect(submit.mock.calls[0][0].zanja).toBeUndefined();
});
it('recepcion con ambas zanjas ocupadas admite espera libre', async () => {
  const submit=vi.fn().mockResolvedValue(undefined);
  render(<ModalCrearOT idMecanicoDefault={0} placaInicial="ABC123" zanjasOcupadas={[1,2]} onClose={vi.fn()} onSubmit={submit}/>);
  const button=screen.getByRole('button',{name:'Crear OT'});
  await waitFor(()=>expect((button as HTMLButtonElement).disabled).toBe(false));
  fireEvent.change(screen.getByPlaceholderText('Ej. 65200'),{target:{value:'50010'}});
  fireEvent.click(button);
  await waitFor(()=>expect(submit).toHaveBeenCalledOnce());
  expect(submit.mock.calls[0][0].zanja).toBeUndefined();
  expect(submit.mock.calls[0][0]).toMatchObject({id_mecanico:0,tipo_aceite:''});
});
it('paginacion navega 20 mas 1', () => {
  const rows = Array.from({length:21},(_,i)=>({...ot,id_ot:i+1,codigo_ot:`OT-NAV-${i+1}`}));
  render(<FichaVehicularHistorial historial={rows} onVerDetalleOT={vi.fn()} />);
  fireEvent.click(screen.getByRole('button',{name:/Siguiente/i}));
  expect(screen.getAllByText(/^OT-NAV-/)).toHaveLength(1);
  expect(screen.getByText('OT-NAV-21')).toBeDefined();
  fireEvent.click(screen.getByRole('button',{name:/Anterior/i}));
  expect(screen.getAllByText(/^OT-NAV-/)).toHaveLength(20);
});
it('paginacion se ajusta cuando se reduce historial', () => {
  const rows = Array.from({length:21},(_,i)=>({...ot,id_ot:i+1,codigo_ot:`OT-RED-${i+1}`}));
  const v=render(<FichaVehicularHistorial historial={rows} onVerDetalleOT={vi.fn()} />);
  fireEvent.click(screen.getByRole('button',{name:/Siguiente/i}));
  v.rerender(<FichaVehicularHistorial historial={[rows[0]]} onVerDetalleOT={vi.fn()} />);
  expect(screen.getByText('OT-RED-1')).toBeDefined();
});
